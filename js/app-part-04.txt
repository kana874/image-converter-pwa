async function extractJpegExifSegment(file){if(!isJpegFile(file))return null;const b=new Uint8Array(await file.arrayBuffer());if(b.length<4||b[0]!==0xff||b[1]!==0xd8)return null;let p=2;while(p+4<=b.length){if(b[p]!==0xff){p++;continue}const marker=b[p+1];if(marker===0xda||marker===0xd9)break;if(marker===0x00||marker===0x01||(marker>=0xd0&&marker<=0xd7)){p+=2;continue}const len=(b[p+2]<<8)|b[p+3];if(len<2||p+2+len>b.length)break;if(marker===0xe1&&len>=8&&b[p+4]===0x45&&b[p+5]===0x78&&b[p+6]===0x69&&b[p+7]===0x66&&b[p+8]===0&&b[p+9]===0){const seg=b.slice(p,p+2+len);normalizeExifOrientation(seg);return seg}p+=2+len}return null}
function normalizeExifOrientation(seg){try{const t=10;if(seg.length<t+8)return;const le=seg[t]===0x49&&seg[t+1]===0x49,be=seg[t]===0x4d&&seg[t+1]===0x4d;if(!le&&!be)return;const u16=o=>le?(seg[o]|seg[o+1]<<8):(seg[o]<<8|seg[o+1]),u32=o=>le?(seg[o]|seg[o+1]<<8|seg[o+2]<<16|seg[o+3]<<24)>>>0:((seg[o]<<24|seg[o+1]<<16|seg[o+2]<<8|seg[o+3])>>>0),ifd=t+u32(t+4);if(ifd+2>seg.length)return;const n=u16(ifd);for(let i=0;i<n;i++){const e=ifd+2+i*12;if(e+12>seg.length)break;if(u16(e)===0x0112&&u16(e+2)===3&&u32(e+4)===1){if(le){seg[e+8]=1;seg[e+9]=0}else{seg[e+8]=0;seg[e+9]=1}return}}}catch(_){}}
async function injectExif(blob,seg){if(!seg)return blob;const out=new Uint8Array(await blob.arrayBuffer());if(out.length<2||out[0]!==0xff||out[1]!==0xd8)return blob;const merged=new Uint8Array(out.length+seg.length);merged.set(out.subarray(0,2),0);merged.set(seg,2);merged.set(out.subarray(2),2+seg.length);return new Blob([merged],{type:"image/jpeg"})}
